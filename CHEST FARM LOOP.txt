    local function isValidKeyFormat(key)
        local pattern = "^[A-Z0-9][A-Z0-9][A-Z0-9][A-Z0-9]%-[A-Z0-9][A-Z0-9][A-Z0-9][A-Z0-9]%-[A-Z0-9][A-Z0-9][A-Z0-9][A-Z0-9]%-[A-Z0-9][A-Z0-9][A-Z0-9][A-Z0-9]$"
        return string.match(key, pattern) ~= nil
    end

    local function verifyKey(key)

    -- ==================== CHEST FARM FUNCTIONS ====================
    local function FindAllChests()
        local chests = {}
        local seen = {}
        
        for _, obj in pairs(Workspace:GetDescendants()) do
            if obj:IsA("Model") then
                local name = obj.Name:lower()
                local keywords = {"chest", "crate", "barrel", "box", "silver", "golden", "diamond", "fragment", "mirage", "treasure"}
                for _, kw in ipairs(keywords) do
                    if name:find(kw) and not seen[obj] then
                        seen[obj] = true
                        table.insert(chests, obj)
                        break
                    end
                end
            end
        end
        
        for _, obj in pairs(Workspace:GetDescendants()) do
            if obj:IsA("BasePart") and obj:FindFirstChild("ClickDetector") then
                local parent = obj.Parent
                if parent and parent:IsA("Model") and not seen[parent] then
                    local name = parent.Name:lower()
                    local keywords = {"chest", "crate", "barrel", "box", "silver", "golden", "diamond", "fragment", "mirage"}
                    for _, kw in ipairs(keywords) do
                        if name:find(kw) then
                            seen[parent] = true
                            table.insert(chests, parent)
                            break
                        end
                    end
                end
            end
        end
        
        return chests
    end

    local function GetChestPart(chest)
        local handle = chest:FindFirstChild("Handle")
        if handle and handle:IsA("BasePart") then return handle end
        for _, part in pairs(chest:GetDescendants()) do
            if part:IsA("BasePart") then return part end
        end
        return nil
    end

    local function GetChestClickDetector(chest)
        for _, obj in pairs(chest:GetDescendants()) do
            if obj:IsA("ClickDetector") then return obj end
        end
        return nil
    end

    local function EnableNoClip()
        if not getgenv().ChestFarm.NoClip then return end
        local char = Player.Character
        if not char then return end
        for _, part in pairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end

    task.spawn(function()
        while true do
            task.wait(0.05)
            if getgenv().ChestFarm.Enabled and getgenv().ChestFarm.NoClip then
                EnableNoClip()
            end
        end
    end)

    local flyBody = nil

    local function FlyToPosition(targetPos)
        if not getgenv().ChestFarm.Enabled then return end
        local char = Player.Character
        if not char then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        
        EnableNoClip()
        
        if not flyBody or not flyBody.Parent then
            flyBody = Instance.new("BodyVelocity")
            flyBody.MaxForce = Vector3.new(1, 1, 1) * 100000
            flyBody.Parent = root
        end
        
        if char:FindFirstChild("Humanoid") then
            char.Humanoid.PlatformStand = true
        end
        
        local speed = getgenv().ChestFarm.FlySpeed or 250
        local distance = (root.Position - targetPos).Magnitude
        local direction = (targetPos - root.Position).Unit
        
        if distance > getgenv().ChestFarm.CollectRadius then
            flyBody.Velocity = direction * speed
            local timeout = 0
            while distance > getgenv().ChestFarm.CollectRadius and getgenv().ChestFarm.Enabled and timeout < 10 do
                distance = (root.Position - targetPos).Magnitude
                direction = (targetPos - root.Position).Unit
                flyBody.Velocity = direction * speed
                if getgenv().ChestFarm.NoClip then EnableNoClip() end
                task.wait(0.03)
                timeout = timeout + 0.03
            end
        end
        
        flyBody.Velocity = Vector3.new(0, 0, 0)
        task.wait(0.05)
    end

    function StopFly()
        if flyBody then
            flyBody.Velocity = Vector3.new(0, 0, 0)
            task.wait(0.1)
            flyBody:Destroy()
            flyBody = nil
        end
        if Player.Character and Player.Character:FindFirstChild("Humanoid") then
            Player.Character.Humanoid.PlatformStand = false
        end
    end

    local function CollectChest(chest)
        if not chest or not chest.Parent then return false end
        local char = Player.Character
        if not char then return false end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return false end
        
        local targetPart = GetChestPart(chest)
        if not targetPart then return false end
        
        local clickDetector = GetChestClickDetector(chest)
        local targetPos = targetPart.Position + Vector3.new(0, 1, 0)
        
        AddConsole("FLYING TO: " .. chest.Name, Color3.fromRGB(0, 200, 255))
        FlyToPosition(targetPos)
        
        if clickDetector then
            pcall(function()
                clickDetector:Click()
                ChestCount = ChestCount + 1
                StatsLabel.Text = "📦 " .. ChestCount
                AddConsole("COLLECTED: " .. chest.Name .. " [" .. ChestCount .. "]", Color3.fromRGB(0, 255, 100))
            end)
        else
            pcall(function()
                targetPart:Click()
                ChestCount = ChestCount + 1
                StatsLabel.Text = "📦 " .. ChestCount
                AddConsole("COLLECTED: " .. chest.Name .. " [" .. ChestCount .. "]", Color3.fromRGB(0, 255, 100))
            end)
        end
        
        task.wait(getgenv().ChestFarm.CollectionDelay or 0.15)
        return true
    end

    function StartFarm()
        getgenv().ChestFarm.Enabled = true
        
        StatusLabel.Text = "⏳ STATUS: SCANNING..."
        StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 0)
        statusDot.BackgroundColor3 = Color3.fromRGB(255, 200, 0)

        AddConsole("NOCLIP: ENABLED", Color3.fromRGB(0, 255, 200))
        AddConsole("----------------------------------------", Color3.fromRGB(100, 100, 100))
        
        if getgenv().ChestFarm.AntiAFK then
            task.spawn(function()
                while getgenv().ChestFarm.Enabled do
                    pcall(function()
                        VirtualUser:CaptureController()
                        VirtualUser:ClickButton2(Vector2.new())
                    end)
                    task.wait(60)
                end
            end)
        end
        
        while getgenv().ChestFarm.Enabled do
            if not Player.Character then
                task.wait(1)
                continue
            end
            
            local chests = FindAllChests()
            
            if #chests == 0 then
                StatusLabel.Text = "⏳ STATUS: NO CHESTS FOUND"
                StatusLabel.TextColor3 = Color3.fromRGB(255, 150, 0)
                AddConsole("SCAN: NO CHESTS DETECTED", Color3.fromRGB(255, 150, 0))
                task.wait(2)
                continue
            end
            
            StatusLabel.Text = "🎯 STATUS: " .. #chests .. " CHESTS FOUND"
            StatusLabel.TextColor3 = Color3.fromRGB(0, 255, 100)
            statusDot.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
            
            local hrp = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                table.sort(chests, function(a, b)
                    local pa = GetChestPart(a)
                    local pb = GetChestPart(b)
                    if not pa then return false end
                    if not pb then return true end
                    return (hrp.Position - pa.Position).Magnitude < (hrp.Position - pb.Position).Magnitude
                end)
            end
            
            for _, chest in ipairs(chests) do
                if not getgenv().ChestFarm.Enabled then break end
                if not chest or not chest.Parent then continue end
                
                pcall(function()
                    CollectChest(chest)
                end)
                
                task.wait(0.3)
            end
            
            AddConsole("CYCLE: COMPLETE - RESTARTING", Color3.fromRGB(200, 200, 200))
            task.wait(1)
        end
        
        StopFly()
        StatusLabel.Text = "⏸ STATUS: OFFLINE"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 0, 0)
        statusDot.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
        StartBtn.Text = "▶ INITIALIZE FARM"
        StartBtn.BorderColor3 = Color3.fromRGB(0, 255, 100)
        StartBtn.TextColor3 = Color3.fromRGB(0, 255, 100)
        
        AddConsole("SYSTEM: FARM TERMINATED", Color3.fromRGB(255, 0, 0))
        AddConsole("----------------------------------------", Color3.fromRGB(100, 100, 100))
    end
